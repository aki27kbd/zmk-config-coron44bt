if(CONFIG_CORON44BT_Z_SHIFT)
  set(z_shift_source ${CMAKE_CURRENT_SOURCE_DIR}/src/behavior_z_shift.c)
  target_sources(app PRIVATE ${z_shift_source})
  set_property(GLOBAL PROPERTY CORON44BT_Z_SHIFT_SOURCE ${z_shift_source})

  # Send Shift (or a pending z) before keymap processes the next position.
  # A nested hold-tap created by tap-dance misses this first interruption.
  function(coron44bt_place_z_shift_listener)
    get_property(listener GLOBAL PROPERTY CORON44BT_Z_SHIFT_SOURCE)
    get_target_property(sources app SOURCES)
    list(REMOVE_ITEM sources ${listener})
    set(index -1)
    set(i 0)
    foreach(source ${sources})
      if(source MATCHES "/keymap\\.c$")
        set(index ${i})
        break()
      endif()
      math(EXPR i "${i} + 1")
    endforeach()
    if(index EQUAL -1)
      message(FATAL_ERROR "coron44bt Z_SHIFT: keymap.c not found")
    endif()
    list(INSERT sources ${index} ${listener})
    set_property(TARGET app PROPERTY SOURCES "${sources}")
    message(STATUS "coron44bt Z_SHIFT: listener before keymap at index ${index}")
  endfunction()
  cmake_language(DEFER DIRECTORY ${CMAKE_SOURCE_DIR} CALL coron44bt_place_z_shift_listener)
endif()
